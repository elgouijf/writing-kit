  \backgroundsetup{contents={}}%
\else%
  \backgroundsetup{scale=1, color=black, opacity=0.04, angle=0,%

% Glossary without an external tool (rule R8): in the text \gterm{key}{Term},
% in back/glossary.tex \gdefn{key}{Term}{definition}.
\newcommand{\gterm}[2]{\hyperlink{gls:#1}{#2}}
\newcommand{\gdefn}[3]{\hypertarget{gls:#1}{}\item[#2] #3}

% One of the two or three statements the document stands on.
\newcommand{\keyclaim}[1]{%
  \par\medskip\noindent
  \fbox{\begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
  \vspace{2pt}\itshape #1\vspace{2pt}\end{minipage}}\par\medskip}

% A gap only the author can fill (rule V13). Printed, never hidden.
\newcommand{\tofill}[1]{%
  \par\medskip\noindent
  \fbox{\begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
  \vspace{2pt}\small\textbf{To be supplied by the author.} #1\vspace{2pt}
  \end{minipage}}\par\medskip}
